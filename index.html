function parseInputContent(rawContent) { let parsedMatches = []; let titleText = "Capechecs - Tournoi FFE"; let roundText = "Résultats de la ronde"; // Nettoyage global du texte const cleanContent = rawContent.replace(/\u00a0/g, ' ').replace(/\r/g, ''); const lines = cleanContent.split('\n').map(l => l.trim()).filter(l => l.length > 0); for (let line of lines) { if (line.includes('Résultats') || line.includes('Round') || line.includes('Ronde')) { roundText = line; continue; } else if (line.includes('Open') || line.includes('Tournoi') || line.includes('Capechecs')) { titleText = line; continue; } // Découpage par tabulation ou par au moins 2 espaces consécutifs (structure classique du presse-papier) let parts = line.split(/\t+|\s{2,}/).map(p => p.trim()).filter(p => p.length > 0); // Si le découpage donne au moins 7 ou 8 colonnes et que la 1re colonne est un numéro d'échiquier if (parts.length >= 7 && !isNaN(parseInt(parts[0]))) { parsedMatches.push({ board: parts[0], whitePts: parts[1] || "", whiteName: parts[2] || "", whiteElo: parts[3] || "", result: parts[4] || "VS", blackName: parts[5] || "", blackElo: parts[6] || "", blackPts: parts[7] || "" }); continue; } // Si le découpage brut a échoué (ex: texte collé d'un seul bloc), fallback sur regex souple const rowRegex = /(\d{1,3})\s+([\d½X\.\-]+)\s+(.+?)\s+(\d{3,4}\s*[FkN]?)\s+(1\s*-\s*0|0\s*-\s*1|½\s*-\s*½|X\s*-\s*X|1\/2\s*-\s*1\/2|1\/2|\s*-\s*|VS)\s+(.+?)\s+(\d{3,4}\s*[FkN]?|EXEMPT|FEXEMPT)\s*([\d½X\.\-]*)/i; const m = line.match(rowRegex); if (m) { parsedMatches.push({ board: m[1].trim(), whitePts: m[2].trim(), whiteName: m[3].trim(), whiteElo: m[4].trim(), result: m[5].trim(), blackName: m[6].trim(), blackElo: m[7].trim(), blackPts: m[8] ? m[8].trim() : "" }); } } // Fallback HTML si le texte brut n'a rien renvoyé if (parsedMatches.length === 0) { const parser = new DOMParser(); const doc = parser.parseFromString(rawContent, 'text/html'); const rows = doc.querySelectorAll('tr'); rows.forEach(row => { const cells = row.querySelectorAll('td'); if (cells.length === 8) { const boardNum = cells[0].textContent.trim(); if (!isNaN(parseInt(boardNum)) && parseInt(boardNum) > 0) { parsedMatches.push({ board: boardNum, whitePts: cells[1].textContent.trim(), whiteName: cells[2].textContent.trim(), whiteElo: cells[3].textContent.trim(), result: cells[4].textContent.trim(), blackName: cells[5].textContent.trim(), blackElo: cells[6].textContent.trim(), blackPts: cells[7].textContent.trim() }); } } }); } // Validation et relance du carrousel if (parsedMatches.length > 0) { document.getElementById('tourneyTitle').textContent = titleText; document.getElementById('roundInfo').textContent = roundText; matchesData = parsedMatches; if (displayTimer) clearInterval(displayTimer); currentIndex = 0; renderNextMatch(); startLoop(); } }
