function parseInputContent(rawContent) { let parsedMatches = []; let titleText = "Capechecs - Tournoi FFE"; let roundText = "Résultats de la ronde"; // 1. Extraction éventuelle du titre / ronde si présent const lines = rawContent.split(/\r?\n/).map(l => l.trim()).filter(l => l.length > 0); for (let line of lines) { if (line.includes('Résultats') || line.includes('Round') || line.includes('Ronde')) { roundText = line; } else if (line.includes('Open') || line.includes('Tournoi') || line.includes('Capechecs')) { titleText = line; } } // 2. Traitement ligne par ligne pour éviter les décalages d'en-tête for (let line of lines) { // Remplace les espaces insécables const cleanLine = line.replace(/\u00a0/g, ' ').trim(); // Cherche une ligne commençant strictement par un numéro d'échiquier (ex: "1 2 RADOMSKY... 1 - 0 MAURIZZI...") const rowRegex = /^(\d{1,3})\s+([\d½X\.\-]+)\s+(.+?)\s+(\d{3,4}\s*[FkN]?)\s+(1\s*-\s*0|0\s*-\s*1|½\s*-\s*½|X\s*-\s*X|1\/2\s*-\s*1\/2|\s*-\s*|VS)\s+(.+?)\s+(\d{3,4}\s*[FkN]?|EXEMPT|FEXEMPT)\s*([\d½X\.\-]*)$/i; const m = cleanLine.match(rowRegex); if (m) { parsedMatches.push({ board: m[1].trim(), whitePts: m[2].trim(), whiteName: m[3].trim(), whiteElo: m[4].trim(), result: m[5].trim(), blackName: m[6].trim(), blackElo: m[7].trim(), blackPts: m[8] ? m[8].trim() : "" }); } } // 3. Si l'analyse ligne par ligne n'a rien donné, tentative via DOM HTML if (parsedMatches.length === 0) { const parser = new DOMParser(); const doc = parser.parseFromString(rawContent, 'text/html'); const rows = doc.querySelectorAll('tr'); rows.forEach(row => { const cells = row.querySelectorAll('td'); if (cells.length === 8) { const boardNum = cells[0].textContent.trim(); if (!isNaN(parseInt(boardNum)) && parseInt(boardNum) > 0) { parsedMatches.push({ board: boardNum, whitePts: cells[1].textContent.trim(), whiteName: cells[2].textContent.trim(), whiteElo: cells[3].textContent.trim(), result: cells[4].textContent.trim(), blackName: cells[5].textContent.trim(), blackElo: cells[6].textContent.trim(), blackPts: cells[7].textContent.trim() }); } } }); } // 4. Validation et affichage if (parsedMatches.length > 0) { document.getElementById('tourneyTitle').textContent = titleText; document.getElementById('roundInfo').textContent = roundText; matchesData = parsedMatches; // Réinitialisation de la boucle sur les nouvelles données if (displayTimer) clearInterval(displayTimer); currentIndex = 0; renderNextMatch(); startLoop(); } }
